Q1.1 INSERT
---------------------------------------------------------------------------
-- 1) Insert a new employee in an existing department (Dno = 2):
INSERT INTO EMPLOYEE (ssn, fname, lname, bdate, address, gender, salary, super_ssn, Dno)
VALUES (107, 'Kareem', 'Adel', '2002-07-19', '33 Maadi, Cairo', 'M', 3900.00, 105, 2);

-- 2) Insert a new project of that department (Dno = 2):
INSERT INTO PROJECT (Pnumber, Pname, Plocation, Dno)
VALUES (50, 'Project Epsilon', 'Cairo', 2);

-- 3) Show both rows with SELECT:
SELECT * FROM EMPLOYEE WHERE ssn = 107;
Output:
ssn | fname  | lname | bdate      | address         | gender | salary  | super_ssn | Dno
107 | Kareem | Adel  | 2002-07-19 | 33 Maadi, Cairo | M      | 3900.00 | 105       | 2

SELECT * FROM PROJECT WHERE Pnumber = 50;
Output:
Pnumber | Pname           | Plocation | Dno
50      | Project Epsilon | Cairo     | 2

-- 4) Try to insert an employee whose Dno does not exist (Dno = 99):
INSERT INTO EMPLOYEE (ssn, fname, lname, bdate, address, gender, salary, super_ssn, Dno)
VALUES (108, 'Ali', 'Hassan', '2001-02-10', '10 Giza St, Giza', 'M', 4000.00, 101, 99);

Output Error Message:
ERROR 1452 (23000): Cannot add or update a child row: a foreign key constraint fails (`company`.`employee`, CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Dno`) REFERENCES `department` (`Dnumber`))
Explanation: MySQL rejects the INSERT because Dno = 99 does not exist in the parent table DEPARTMENT(Dnumber).

Explanation: MySQL blocks the UPDATE because Dno = 99 violates the foreign key constraint referencing DEPARTMENT(Dnumber).

Q1.3 DELETE
---------------------------------------------------------------------------
-- (a) Delete one dependent and one WORKS_ON row:
DELETE FROM DEPENDENT WHERE essn = 105 AND dependent_name = 'Salma';
DELETE FROM WORKS_ON WHERE essn = 106 AND Pno = 30;
Output:
Query OK, 1 row affected (for each DELETE).

-- (b) Try to delete a department that still has employees (Dnumber = 2):
DELETE FROM DEPARTMENT WHERE Dnumber = 2;

Output Error Message:
ERROR 1451 (23000): Cannot delete or update a parent row: a foreign key constraint fails (`company`.`employee`, CONSTRAINT `employee_ibfk_1` FOREIGN KEY (`Dno`) REFERENCES `department` (`Dnumber`))
Explanation & Removal Order: MySQL prevents deleting Department 2 while employees/projects still reference it. To delete Department 2 safely, you must first remove or reassign dependent child rows in order:
1) Delete rows in DEPENDENT and WORKS_ON for employees of Dno = 2 (or reassign/delete those employees).
2) Delete or reassign rows in PROJECT where Dno = 2 (and any WORKS_ON rows referencing those projects).
3) Update/remove any super_ssn references pointing to employees of Dno = 2, then delete those EMPLOYEE rows.
4) Finally, execute DELETE FROM DEPARTMENT WHERE Dnumber = 2;.

-- (c) Transaction test with DELETE without WHERE and ROLLBACK:
START TRANSACTION;
DELETE FROM DEPENDENT;
SELECT * FROM DEPENDENT;
-- Output after DELETE: Empty set (0 rows)
ROLLBACK;
SELECT * FROM DEPENDENT;
-- Output after ROLLBACK:
essn | dependent_name | gender | bdate      | relationship
101  | Nour           | F      | 2018-08-22 | Daughter
101  | Youssef        | M      | 2015-04-10 | Son
Explanation: DELETE without WHERE removed all remaining rows in DEPENDENT inside the active transaction, and ROLLBACK undid the uncommitted deletion and restored those rows.

---------------------------------------------------------------------------
-- (a) Employees earning between 3000 and 6000 living in Alexandria, highest salary first:
SELECT fname, lname, salary
FROM EMPLOYEE

Q1.7 HAVING
---------------------------------------------------------------------------
-- (a) Departments that have more than 2 employees:
SELECT Dno, COUNT(*) AS num_employees
FROM EMPLOYEE
GROUP BY Dno
HAVING COUNT(*) > 2;

Output:
Dno | num_employees
1   | 4
2   | 3

-- (b) Departments with more than 2 employees whose salary is above 3000:
SELECT Dno, COUNT(*) AS high_paid_employees
FROM EMPLOYEE
WHERE salary > 3000
GROUP BY Dno
HAVING COUNT(*) > 2;

Output:
Dno | high_paid_employees
1   | 3
2   | 3

Explanation: WHERE filters individual rows before grouping (and cannot use aggregate functions), whereas HAVING filters entire groups after GROUP BY has been computed (and can use aggregate functions like COUNT(*)).

Q1.10 LEFT JOIN
---------------------------------------------------------------------------
-- (a) Every department with its number of employees, including departments with 0:
SELECT d.Dnumber, d.Dname, COUNT(e.ssn) AS employee_count
FROM DEPARTMENT d
LEFT JOIN EMPLOYEE e ON d.Dnumber = e.Dno
GROUP BY d.Dnumber, d.Dname;

Output:
Dnumber | Dname          | employee_count
1       | Research       | 4
2       | Administration | 3
3       | Headquarters   | 0

-- (b) Projects that nobody works on:
SELECT p.Pnumber, p.Pname, p.Plocation
FROM PROJECT p
LEFT JOIN WORKS_ON w ON p.Pnumber = w.Pno
WHERE w.essn IS NULL;

Output:
Pnumber | Pname           | Plocation
40      | Project Delta   | Alexandria
50      | Project Epsilon | Cairo

Q1.12 JOIN + GROUP BY + HAVING
---------------------------------------------------------------------------
SELECT p.Pname, SUM(w.hours) AS total_hours
FROM PROJECT p
JOIN WORKS_ON w ON p.Pnumber = w.Pno
GROUP BY p.Pnumber, p.Pname
HAVING SUM(w.hours) > 20
ORDER BY total_hours DESC;

Output:
Pname         | total_hours
Project Alpha | 41.50
Project Beta  | 29.50
